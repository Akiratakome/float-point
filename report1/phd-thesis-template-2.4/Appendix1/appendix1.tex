\begin{figure}[p]
\centering
\includegraphics[width=\textwidth]{Figs/report1/appendix_lw3_self_convergence}
\caption[LW3 fp64 self-convergence against finer grids]{LW3 fp64 self-convergence: density $L_1$ between successive grid pairs ($400$ vs $800$, $800$ vs $1600$), on a log--log axis. The fitted slope gives an observed self-convergence order of about $0.47$, in line with the sub-quadratic behaviour seen for the 1D shock tubes in Table~\ref{tab:ch5-1d-exact-convergence}. This supports treating $1600^2$ fp64 as a finer-grid numerical reference for LW3 rather than an exact-solution comparator.}
\label{fig:app-lw3-self-convergence}
\end{figure}
